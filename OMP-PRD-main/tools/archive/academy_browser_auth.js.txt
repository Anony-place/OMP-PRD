async function makePage(options = {}) {
  const context = await browser.newContext({
    viewport: { width: 1440, height: 1000 },
    colorScheme: options.theme || "dark",
    javaScriptEnabled: !options.noJavaScript,
  });
  await context.addInitScript(
    (f) => {
      window.__academyFixture = f;
    },
    { ...fixture, ...options.multiSubject, ...options },
  );
  await context.route("https://esm.sh/**", (route) =>
    options.offline
      ? route.abort()
      : route.fulfill({ contentType: "text/javascript", body: mockModule }),
  );
  await context.route("**/assets/data/academy-catalogue.json", (route) =>
    route.fulfill({
      json: { ...fixture, ...options.multiSubject, ...options },
    }),
  );
  if (options.verification)
    await context.route(
      "**/functions/v1/learning-certificate-verify?**",
      (route) =>
        options.verification === "network"
          ? route.abort()
          : options.verification === "timeout"
            ? new Promise(() => {})
            : route.fulfill(options.verification),
    );
  if (options.multiSubject) {
    const { curriculum, resolvePage } =
      await import("../assets/js/academy-model.mjs");
    const { publicView, workspaceView, esc } =
      await import("../assets/js/academy-views.mjs");
    const model = curriculum(options.multiSubject);
    const template = fs.readFileSync(
      path.join(ROOT, "tools/academy-shell.html"),
      "utf8",
    );
    await context.route(
      (url) =>
        url.origin === new URL(BASE).origin &&
        url.pathname.startsWith("/learning"),
      (route) => {
        if (route.request().resourceType() !== "document")
          return route.continue();
        const page = resolvePage(
          model,
          new URL(route.request().url()).pathname,
        );
        if (["notfound", "redirect"].includes(page.type))
          return route.continue();
        const html = template
          .replace(
            "%%ACADEMY_CONTENT%%",
            workspaceView(model, page, publicView(model, page)),
          )
          .replace(
            /<title>[\s\S]*?<\/title>/,
            `<title>${esc(page.title)} | OMP Academy</title>`,
          )
          .replace(
            /<link\s+rel="canonical"[^>]*>/,
            `<link rel="canonical" href="https://oopsmypassword.web.app${page.url}">`,
          );
        return route.fulfill({ contentType: "text/html", body: html });
      },
    );
  }
  const page = await context.newPage();
  page.on("pageerror", (e) => errors.push(e.message));
  const response = await page.goto(BASE + (options.route || "/learning"));
  if (response.status() !== 404 && !options.noJavaScript)
    await page.waitForSelector(
      options.route?.startsWith("/certificate")
        ? ".academy-empty, .certificate-shell"
        : "#academy-account",
      { state: "attached" },
    );
  return page;
}

async function noOverflow(page) {
  assert.equal(
    await page.evaluate(
      () => document.documentElement.scrollWidth > innerWidth,
    ),
    false,
  );
}
async function audit(page, name) {
  const result = await new AxeBuilder({ page })
    .withTags(["wcag2a", "wcag2aa", "wcag21aa"])
    .analyze();
  assert.deepEqual(
    result.violations.map((v) => ({
      id: v.id,
      nodes: v.nodes.map((n) => n.target),
    })),
    [],
    name,
  );
  ok(name);
}
